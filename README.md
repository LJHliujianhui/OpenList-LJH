<!-- 音乐增强【优化修复版】适配Aplayer -->
<script>
  window.initMusicPlus = false;
  // 简易DOM选择器（避免依赖jQuery）
  window.$q = window.$q || function(val) { return document.querySelector(val) }
  window.$qa = window.$qa || function(val) { return Array.from(document.querySelectorAll(val)) }

  // 生成随机渐变背景
  function generateRandomGradient() {
    const randomColor = () => `rgba(${Math.floor(Math.random() * 255)}, ${Math.floor(Math.random() * 255)}, ${Math.floor(Math.random() * 255)}, ${(Math.random()*0.4 + 0.15).toFixed(2)})`;
    const color1 = randomColor();
    const color2 = randomColor();
    return `linear-gradient(to right, ${color1}, ${color2})`;
  }

  // 暗黑模式/屏幕宽度适配（修复：移除jQuery，原生JS实现）
  function darkUIChange() {
    const documentWidth = window.innerWidth;
    const isMobile = documentWidth < 600;
    const isDarkUI = !!$q('.hope-ui-dark');
    const lrcBox = $q('.music-plus .aplayer .aplayer-lrc');
    if(!lrcBox) return;

    if (isDarkUI) {
      lrcBox.style.top = isMobile ? "332px" : "315px";
    } else {
      // 浅色模式默认位置
      lrcBox.style.top = window.innerWidth >= 1200 ? "212px" : "180px";
    }
  }

  // 全局注入样式
  ;(function(){
    const style = document.createElement('style')
    style.innerHTML = `
      .music-plus .aplayer .aplayer-lrc { 
        position: fixed; 
        top: 180px; 
        left: 0; 
        right: 0; 
        width: 88%; 
        height: 100px; 
        max-width: 1000px; 
        margin: 0 auto; 
        padding: 50px 0; 
        transform: scale(1.3);
        z-index: 999; 
        backdrop-filter: blur(10px);
        -webkit-backdrop-filter: blur(10px);
        box-shadow: 0 4px 8px rgba(0,0,0,0.5);
        transition: background 0.5s ease, top 0.3s ease;
      }
      .music-plus .aplayer .aplayer-lrc-current { 
        font-weight: bold; 
        transform: scale(0.95) !important; 
      }
      .music-plus .aplayer .aplayer-lrc p { 
        color: #fff;
        opacity: 1;
        transform: scale(0.9); 
      }
      .music-control { 
        position: absolute; 
        right: 4px; 
        top: 4px; 
        display: flex; 
        align-items: center; 
      }
      .music-control i { 
        cursor: pointer; 
        width: 22px; 
        height: 22px; 
        display: flex; 
        align-items: center; 
        justify-content: center; 
        font-size: 14px; 
        border-radius: 50%; 
        background: rgba(255, 255, 255, 0.2);
        margin-left: 4px; 
        font-style: normal; 
      }
      /*电脑大屏适配*/
      @media screen and (min-width: 1200px) {
        .music-plus .aplayer .aplayer-lrc { 
          width: 410px; 
          height: 100px; 
          padding: 100px 0; 
          top: 212px;
        }
      }
    `
    document.head.appendChild(style)
  })();

  // 延迟创建【音乐加强】按钮 + 绑定快捷键
  setTimeout(() => {
    const stack = $q('#musicPlus2')
    if(!stack) return console.warn("未找到 #musicPlus2 挂载容器");

    const separator = document.createElement('span')
    separator.textContent = '|'
    stack.appendChild(separator)

    const toggleBtn = document.createElement('a')
    toggleBtn.style = `padding: var(--hope-space-1_5); cursor: pointer;`
    toggleBtn.textContent = '音乐加强'
    toggleBtn.onclick = changeMusic
    stack.appendChild(toggleBtn)

    // 键盘快捷键监听
    document.addEventListener('keydown', function(e) {
      if (!e.ctrlKey || !window.initMusicPlus) return;
      let type = null;
      if (['4','['].includes(e.key)) type = 1;    // 上一曲
      if (['6',']'].includes(e.key)) type = 3;    // 下一曲
      if (['5','\\'].includes(e.key)) type = 2;   // 暂停/播放
      if (type) {
        e.preventDefault()
        e.stopPropagation()
        musicMove(type)
      }
    })
  }, 1200)

  // 开关：开启/关闭音乐增强模式
  function changeMusic() {
    const lrcWarp = $q('.aplayer .aplayer-lrc')
    if (!lrcWarp) {
      console.warn("Aplayer歌词容器未加载，无法开启音乐增强");
      return;
    }

    // 生成控制按钮
    if (!window.initMusicPlus || !$q('.music-control')) {
      const control = document.createElement('div')
      control.className = 'music-control'
      control.innerHTML = `
        <i onclick="musicMove(1)" title="Ctrl + [ 上一曲">◄</i>
        <i onclick="musicMove(2)" title="Ctrl + \ 暂停/播放">◼</i>
        <i onclick="musicMove(3)" title="Ctrl + ] 下一曲">►</i>
      `
      $q('.aplayer-body')?.append(control)
    }

    window.initMusicPlus = true
    // 切换body class开关效果
    document.body.classList.toggle('music-plus')
    
    // 切换后分配随机渐变 + 适配位置
    if(document.body.classList.contains('music-plus')){
      lrcWarp.style.background = generateRandomGradient();
    }
    darkUIChange();
  }

  // 播放器控制逻辑：上一曲/下一曲/暂停播放
  function musicMove(type) {
    const coverImg = $q('.aplayer-pic');
    if(type === 2 && coverImg){
      coverImg.click();
      return;
    }
    const list = $qa('.aplayer-list li')
    if(list.length === 0) return;
    const curIndex = list.findIndex(item => item.classList.contains('aplayer-list-light'))
    let targetIndex = curIndex;
    if(type === 1) targetIndex = curIndex - 1;
    if(type === 3) targetIndex = curIndex + 1;
    // 循环边界
    if(targetIndex < 0) targetIndex = list.length - 1;
    if(targetIndex >= list.length) targetIndex = 0;
    list[targetIndex].click()
  }
</script>
